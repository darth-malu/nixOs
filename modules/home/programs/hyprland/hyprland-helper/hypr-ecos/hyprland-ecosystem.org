* Hyprsunset
#+begin_src nix :tangle hyprsunset.nix
{
  services.hyprsunset = {
    enable = true;
    transitions = {
      sunrise = {
        calendar = "*-*-* 06:00:00"; # TODO systemd time?
        requests = [
          [
            "temperature"
            "6500"
          ]
          [ "gamma 100" ]
        ];
      };
      sunset = {
        calendar = "*-*-* 19:00:00";
        requests = [
          [
            "temperature"
            "3500"
          ]
        ];
      };
    };
  };
}
#+end_src

# Enable blue-light filter
#+begin_src sh
hyprctl hyprsunset temperature 2500
Disable blue-light filter
hyprctl hyprsunset identity
#+end_src

#+begin_src sh
# Set gamma to 50%
hyprctl hyprsunset gamma 50
# Increase gamma by 10%
hyprctl hyprsunset gamma +10
#+end_src
* Hypridle
#+begin_src nix :tangle hypridle.nix
{ lib, config, ... }:
{
  options.hypridle = {
    enable = lib.mkEnableOption "hypridle";
  };
  config = lib.mkIf config.hypridle.enable {
    services.hypridle = {
      enable = true;
      settings = {
        general = {
          after_sleep_cmd = "hyprctl dispatch dpms on";
          # ignore_dbus_inhibit = false;
          lock_cmd = "hyprlock";
        };
        listener = [
          {
            timeout = 240;
            on-timeout = "hyprlock";
          }
          {
            timeout = 500;
            on-timeout = "hyprctl dispatch dpms off";
            on-resume = "hyprctl dispatch dpms on";
          }
        ];
      };
    };
  };
}
